def parse_duration(s):
    """Parse strings like "90s", "15m", "2h" into seconds."""
    units = {"s": 1, "m": 60, "h": 3600}
    s = s.strip().lower()
    if not s or s[-1] not in units:
        raise ValueError(f"bad duration: {s!r}")
    return int(s[:-1]) * units[s[-1]]
